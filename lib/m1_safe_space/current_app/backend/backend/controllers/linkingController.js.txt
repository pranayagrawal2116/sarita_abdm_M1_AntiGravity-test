const axios = require("axios");
const { v4: uuidv4 } = require("uuid");
const { getGatewayToken } = require("../services/gatewayService");
const { getHeaders } = require("../utils/headers");
const {
    updateFlow,
    getFlow,
    findFlowByRequestId,
    findFlowByTransactionId,
} = require("../utils/linkingStore");

const getErrorPayload = (err) => {
    const status = err.response?.status || 500;
    const data = err.response?.data;

    if (typeof data === "string") {
        return { status, body: { error: data } };
    }

    if (data && typeof data === "object") {
        return { status, body: data };
    }

    return {
        status,
        body: { error: err.message || "Unknown ABDM linking API error" },
    };
};

const toText = (value) => {
    if (value === null || value === undefined) return "";
    const text = String(value).trim();
    return text.length > 0 ? text : "";
};

const getByPath = (source, path) => {
    let current = source;
    for (const key of path) {
        if (Array.isArray(current) && typeof key === "number") {
            current = current[key];
            continue;
        }

        if (current && typeof current === "object" && key in current) {
            current = current[key];
            continue;
        }

        return undefined;
    }
    return current;
};

const findFirstString = (source, paths) => {
    for (const path of paths) {
        const text = toText(getByPath(source, path));
        if (text) return text;
    }
    return "";
};

const getHiuId = () =>
    toText(process.env.HIU_ID) ||
    toText(process.env.ABDM_CLIENT_ID);

const formatXAuthToken = (token) =>
    token.startsWith("Bearer ") ? token : `Bearer ${token}`;

const getLinkingHeaders = (gatewayToken, phrToken) => {
    const hiuId = getHiuId();
    if (!hiuId) {
        throw new Error("Missing HIU_ID or ABDM_CLIENT_ID configuration");
    }

    return {
        ...getHeaders(gatewayToken),
        "X-AUTH-TOKEN": formatXAuthToken(phrToken),
        "X-HIU-ID": hiuId,
    };
};

const mapIdentifierType = (value) => {
    const text = toText(value).toUpperCase();
    switch (text) {
        case "Abha_ADDRESS":
        case "Abha NUMBER":
        case "Abha_ADDRESS ":
            return "Abha_ADDRESS";
        case "Abha_NUMBER":
            return "Abha_NUMBER";
        case "MOBILE":
        case "MOBILE_NUMBER":
            return "MOBILE";
        case "AADHAAR":
        case "AADHAAR_NUMBER":
            return "AADHAAR_NUMBER";
        default:
            return text;
    }
};

const normalizeCareContexts = (value) => {
    if (!Array.isArray(value)) return [];
    return value
        .map((item) => ({
            referenceNumber: findFirstString(item, [["referenceNumber"], ["reference"], ["id"]]),
            display: findFirstString(item, [["display"], ["name"]]),
        }))
        .filter((item) => item.referenceNumber);
};

const normalizeDiscoveredPatients = (payload) => {
    const list =
        getByPath(payload, ["patient"]) ||
        getByPath(payload, ["patients"]) ||
        getByPath(payload, ["discovery", "patient"]) ||
        [];

    if (!Array.isArray(list)) return [];

    return list
        .map((item) => {
            const hiTypeValue =
                getByPath(item, ["hiType"]) ??
                getByPath(item, ["hiTypes"]) ??
                getByPath(item, ["careContexts", 0, "hiType"]);

            const hiTypes = Array.isArray(hiTypeValue)
                ? hiTypeValue.map((entry) => toText(entry)).filter(Boolean)
                : [toText(hiTypeValue)].filter(Boolean);

            return {
                referenceNumber: findFirstString(item, [["referenceNumber"], ["id"]]),
                display: findFirstString(item, [["display"], ["name"]]),
                hiTypes,
                count: Number(item.count || normalizeCareContexts(item.careContexts).length || 0),
                careContexts: normalizeCareContexts(item.careContexts),
                raw: item,
            };
        })
        .filter((item) => item.referenceNumber || item.careContexts.length > 0);
};

const summarizeFlow = (flow) => ({
    flowId: flow.flowId,
    stage: flow.stage || "PENDING",
    transactionId: flow.transactionId || "",
    hipId: flow.hipId || "",
    identifierType: flow.identifierType || "",
    identifierValue: flow.identifierValue || "",
    error: flow.error || "",
    discoveredPatients: flow.discoveredPatients || [],
    link: flow.link || null,
    confirmedPatients: flow.confirmedPatients || [],
    updatedAt: flow.updatedAt,
});

const normalizeProviderList = (payload) => {
    const list = Array.isArray(payload)
        ? payload
        : getByPath(payload, ["providers"]) ||
          getByPath(payload, ["results"]) ||
          getByPath(payload, ["response"]) ||
          [];

    if (!Array.isArray(list)) return [];

    return list
        .map((item) => {
            const hipId = findFirstString(item, [
                ["identifier", "id"],
                ["id"],
                ["hipId"],
                ["providerId"],
            ]);

            return {
                hipId,
                name:
                    findFirstString(item, [
                        ["identifier", "name"],
                        ["name"],
                        ["display"],
                    ]) || hipId,
                facilityType: findFirstString(item, [["facilityType"], ["type"]]),
                city: findFirstString(item, [["city"], ["address", "city"]]),
                stateCode: findFirstString(item, [["stateCode"], ["address", "stateCode"]]),
                isHip: Boolean(item?.isHIP ?? item?.isHip ?? true),
                raw: item,
            };
        })
        .filter((item) => item.hipId);
};

const resolveFlowFromCallback = (body) => {
    const requestId = findFirstString(body, [
        ["response", "requestId"],
        ["requestId"],
    ]);
    const transactionId = findFirstString(body, [["transactionId"]]);

    return (
        findFlowByRequestId(requestId) ||
        findFlowByTransactionId(transactionId)
    );
};

exports.discoverCareContexts = async (req, res) => {
    try {
        const phrToken = toText(req.header("X-Auth-Token"));
        const hipId = toText(req.body?.hipId);
        const identifierType = mapIdentifierType(req.body?.identifierType);
        const identifierValue = toText(req.body?.identifierValue);

        if (!phrToken) {
            return res.status(401).json({ error: "Missing login session. Please log in again." });
        }

        if (!hipId) {
            return res.status(400).json({ error: "Health facility HIP ID is required" });
        }

        if (!identifierType || !identifierValue) {
            return res.status(400).json({ error: "Patient identifier is required" });
        }

        const gatewayToken = await getGatewayToken();
        const headers = getLinkingHeaders(gatewayToken, phrToken);
        const flowId = uuidv4();

        updateFlow(flowId, {
            stage: "DISCOVER_REQUESTED",
            discoveryRequestId: headers["REQUEST-ID"],
            hipId,
            identifierType,
            identifierValue,
            phrAddress: toText(req.body?.phrAddress),
            discoveredPatients: [],
            confirmedPatients: [],
            link: null,
            error: "",
        });

        const response = await axios.post(
            `${process.env.GATEWAY_BASE}/api/hiecm/user-initiated-linking/v3/patient/care-context/discover`,
            {
                hipId,
                unverifiedIdentifiers: [
                    {
                        type: identifierType,
                        value: identifierValue,
                    },
                ],
            },
            { headers }
        );

        const responseTransactionId = findFirstString(response.data || {}, [["transactionId"]]);

        const flow = updateFlow(flowId, {
            transactionId: responseTransactionId,
            response: response.data || {},
        });

        return res.json(summarizeFlow(flow));
    } catch (err) {
        const { status, body } = getErrorPayload(err);
        res.status(status).json(body);
    }
};

exports.getLinkingStatus = async (req, res) => {
    const flow = getFlow(String(req.params?.flowId || "").trim());
    if (!flow) {
        return res.status(404).json({ error: "Verification flow not found" });
    }

    return res.json(summarizeFlow(flow));
};

exports.initLinking = async (req, res) => {
    try {
        const phrToken = toText(req.header("X-Auth-Token"));
        const flowId = toText(req.body?.flowId);

        if (!phrToken) {
            return res.status(401).json({ error: "Missing login session. Please log in again." });
        }

        if (!flowId) {
            return res.status(400).json({ error: "Flow ID is required" });
        }

        const flow = getFlow(flowId);
        if (!flow) {
            return res.status(404).json({ error: "Verification flow not found" });
        }

        if (!flow.transactionId) {
            return res.status(400).json({
                error: "Discovery is still pending. Please wait for the ABDM callback.",
            });
        }

        const selectedReferences = Array.isArray(req.body?.selectedReferenceNumbers)
            ? req.body.selectedReferenceNumbers.map((item) => toText(item)).filter(Boolean)
            : [];

        const patients = (flow.discoveredPatients || [])
            .filter(
                (patient) =>
                    selectedReferences.length === 0 ||
                    selectedReferences.includes(toText(patient.referenceNumber))
            )
            .map((patient) => ({
                referenceNumber: patient.referenceNumber,
                display: patient.display,
                careContexts: (patient.careContexts || []).map((careContext) => ({
                    referenceNumber: careContext.referenceNumber,
                    display: careContext.display,
                })),
                hiType:
                    patient.hiTypes?.length === 1
                        ? patient.hiTypes[0]
                        : patient.hiTypes,
                count: patient.count || patient.careContexts?.length || 0,
            }))
            .filter((patient) => patient.referenceNumber && (patient.careContexts || []).length > 0);

        if (patients.length === 0) {
            return res.status(400).json({
                error: "At least one discovered care context is required to continue",
            });
        }

        const gatewayToken = await getGatewayToken();
        const headers = getLinkingHeaders(gatewayToken, phrToken);

        updateFlow(flowId, {
            stage: "LINK_INIT_REQUESTED",
            initRequestId: headers["REQUEST-ID"],
            error: "",
            selectedPatients: patients,
        });

        const response = await axios.post(
            `${process.env.GATEWAY_BASE}/api/hiecm/user-initiated-linking/v3/link/care-context/init`,
            {
                transactionId: flow.transactionId,
                patient: patients,
            },
            { headers }
        );

        const nextFlow = updateFlow(flowId, {
            response: response.data || {},
        });

        return res.json(summarizeFlow(nextFlow));
    } catch (err) {
        const { status, body } = getErrorPayload(err);
        res.status(status).json(body);
    }
};

exports.confirmLinking = async (req, res) => {
    try {
        const phrToken = toText(req.header("X-Auth-Token"));
        const flowId = toText(req.body?.flowId);
        const otp = toText(req.body?.otp);

        if (!phrToken) {
            return res.status(401).json({ error: "Missing login session. Please log in again." });
        }

        if (!flowId) {
            return res.status(400).json({ error: "Flow ID is required" });
        }

        if (!/^\d{4,8}$/.test(otp)) {
            return res.status(400).json({ error: "OTP is required" });
        }

        const flow = getFlow(flowId);
        if (!flow) {
            return res.status(404).json({ error: "Verification flow not found" });
        }

        const linkRefNumber = findFirstString(flow, [
            ["link", "referenceNumber"],
            ["initRaw", "link", "referenceNumber"],
            ["response", "link", "referenceNumber"],
        ]);

        if (!linkRefNumber) {
            return res.status(400).json({
                error: "Link reference number is not available yet. Please wait for the OTP step.",
            });
        }

        const gatewayToken = await getGatewayToken();
        const headers = getLinkingHeaders(gatewayToken, phrToken);

        updateFlow(flowId, {
            stage: "LINK_CONFIRM_REQUESTED",
            confirmRequestId: headers["REQUEST-ID"],
            error: "",
        });

        const response = await axios.post(
            `${process.env.GATEWAY_BASE}/api/hiecm/user-initiated-linking/v3/link/care-context/confirm`,
            {
                token: otp,
                linkRefNumber,
            },
            { headers }
        );

        const nextFlow = updateFlow(flowId, {
            response: response.data || {},
        });

        return res.json(summarizeFlow(nextFlow));
    } catch (err) {
        const { status, body } = getErrorPayload(err);
        res.status(status).json(body);
    }
};

exports.searchProviders = async (req, res) => {
    try {
        const name = toText(req.query?.name);

        if (!name) {
            return res.status(400).json({ error: "Facility name is required" });
        }

        const gatewayToken = await getGatewayToken();
        const headers = getHeaders(gatewayToken);
        const response = await axios.get(
            `${process.env.GATEWAY_BASE}/api/hiecm/gateway/v3/providers`,
            {
                headers,
                params: {
                    stateCode: req.query?.stateCode || -1,
                    districtCode: req.query?.districtCode || -1,
                    name,
                },
            }
        );

        return res.json({
            providers: normalizeProviderList(response.data),
        });
    } catch (err) {
        const { status, body } = getErrorPayload(err);
        res.status(status).json(body);
    }
};

exports.getProviderById = async (req, res) => {
    try {
        const providerId = toText(req.params?.providerId);

        if (!providerId) {
            return res.status(400).json({ error: "Provider ID is required" });
        }

        const gatewayToken = await getGatewayToken();
        const headers = getHeaders(gatewayToken);
        const response = await axios.get(
            `${process.env.GATEWAY_BASE}/api/hiecm/gateway/v3/providers/${encodeURIComponent(providerId)}`,
            { headers }
        );

        const providers = normalizeProviderList([response.data]);
        return res.json({
            provider: providers[0] || null,
        });
    } catch (err) {
        const { status, body } = getErrorPayload(err);
        res.status(status).json(body);
    }
};

exports.onDiscover = async (req, res) => {
    const flow = resolveFlowFromCallback(req.body || {});
    if (flow) {
        updateFlow(flow.flowId, {
            stage: req.body?.error ? "FAILED" : "DISCOVERED",
            transactionId: findFirstString(req.body, [["transactionId"]]) || flow.transactionId,
            discoveredPatients: normalizeDiscoveredPatients(req.body),
            discoveryRaw: req.body,
            error: toText(req.body?.error?.message) || "",
        });
    }

    return res.status(202).json({});
};

exports.onInit = async (req, res) => {
    const flow = resolveFlowFromCallback(req.body || {});
    if (flow) {
        updateFlow(flow.flowId, {
            stage: req.body?.error ? "FAILED" : "OTP_PENDING",
            transactionId: findFirstString(req.body, [["transactionId"]]) || flow.transactionId,
            link: getByPath(req.body, ["link"]) || null,
            initRaw: req.body,
            error: toText(req.body?.error?.message) || "",
        });
    }

    return res.status(202).json({});
};

exports.onConfirm = async (req, res) => {
    const flow = resolveFlowFromCallback(req.body || {});
    if (flow) {
        updateFlow(flow.flowId, {
            stage: req.body?.error ? "FAILED" : "LINKED",
            confirmedPatients: normalizeDiscoveredPatients(req.body),
            confirmRaw: req.body,
            error: toText(req.body?.error?.message) || "",
        });
    }

    return res.status(202).json({});
};
