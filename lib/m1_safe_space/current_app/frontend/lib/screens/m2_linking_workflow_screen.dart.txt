  dynamic _byPath(dynamic source, List<dynamic> path) {
    dynamic current = source;
    for (final segment in path) {
      if (current is Map<String, dynamic> && segment is String) {
        current = current[segment];
        continue;
      }
      if (current is List && segment is int) {
        if (segment < 0 || segment >= current.length) {
          return null;
        }
        current = current[segment];
        continue;
      }
      return null;
    }
    return current;
  }

  String _firstNonEmpty(dynamic source, List<List<dynamic>> paths) {
    for (final path in paths) {
      final value = _byPath(source, path);
      final text = value?.toString().trim() ?? '';
      if (text.isNotEmpty) return text;
    }
    return '';
  }

  Future<Map<String, dynamic>> _waitForGenerateTokenCallback(
    String requestId,
  ) async {
    final deadline = DateTime.now().add(_apiTimeout);
    Object? lastError;
    bool callbackNeverReachedBackend = false;

    while (DateTime.now().isBefore(deadline)) {
      try {
        final callback = await HipLinkingApiService.fetchTokenCallback(
          requestId,
        );
        if (_extractLinkingToken(callback).isNotEmpty) {
          return callback;
        }
        lastError = Exception(
          'Token callback reached the app but did not contain linkToken.',
        );
      } catch (error) {
        final message = error.toString().toLowerCase();
        if (message.contains('callback not received yet') ||
            message.contains('(404)')) {
          callbackNeverReachedBackend = true;
        } else {
          lastError = error;
        }
      }

      await Future.delayed(_callbackPollInterval);
    }

    throw TimeoutException(
      callbackNeverReachedBackend
          ? 'ABDM did not send the token callback to this app within ${_apiTimeout.inSeconds} seconds for requestId $requestId. Check the public callback URL/tunnel: ${_tokenCallbackUrl.isNotEmpty
                ? _tokenCallbackUrl
                : _callbackPublicBaseUrl.isNotEmpty
                ? _callbackPublicBaseUrl
                : 'not configured'}.'
          : lastError?.toString() ??
                'ABDM callback did not return a link token within ${_apiTimeout.inSeconds} seconds',
    );
  }

  Future<bool> _execute({
    required String action,
    required Future<Map<String, dynamic>> Function() call,
    Duration? timeout,
  }) async {
    setState(() => _callingApi = true);
    try {
      final response = await call().timeout(
        timeout ?? _apiTimeout,
        onTimeout: () => throw TimeoutException(
          "ABDM API timed out after ${(timeout ?? _apiTimeout).inSeconds} seconds",
        ),
      );
      if (!mounted) return false;
      _applyAutoFillFromResponse(response);
      setState(() {
        _lastAction = action;
        _lastResponse = _pretty(response);
      });
      return true;
    } catch (e) {
      if (!mounted) return false;
      final message = _errorMessage(e);
      setState(() {
        _lastAction = '$action (error)';
        _lastResponse = message;
      });
      await _showErrorDialog(title: 'API Request Failed', error: e);
      return false;
    } finally {
      if (mounted) {
        setState(() => _callingApi = false);
      }
    }
  }

  Future<void> _callGenerateToken() async {
    final hipId = _required(_hipIdController, 'HIP ID');
    final AbhaNumber = _required(_AbhaNumberController, 'ABHA Number');
    final AbhaAddress = _required(_AbhaAddressController, 'ABHA Address');
    final patientName = _required(_patientNameController, 'Patient Name');
    final patientGender = _required(_patientGenderController, 'Patient Gender');
    final patientYearOfBirth = _required(
      _patientYearOfBirthController,
      'Year Of Birth',
    );

    final success = await _execute(
      action: 'POST /hip/link/token/generate',
      timeout: const Duration(seconds: 90),
      call: () async {
        final response = await HipLinkingApiService.generateToken({
          'hipId': hipId,
          'AbhaNumber': AbhaNumber,
          'AbhaAddress': AbhaAddress,
          'name': patientName,
          'gender': patientGender,
          'yearOfBirth': patientYearOfBirth,
        });

        if (_extractLinkingToken(response).isNotEmpty) {
          return response;
        }

        final requestId = _extractRequestId(response);
        if (requestId.isEmpty) {
          return response;
        }

        final callback = await _waitForGenerateTokenCallback(requestId);
        return {
          ...response,
          'callbackReceived': true,
          'callback': callback,
          'linkToken': _extractLinkingToken(callback).isNotEmpty
              ? _extractLinkingToken(callback)
              : _extractLinkingToken(response),
          'response': {
            ...((response['response'] is Map<String, dynamic>)
                ? response['response'] as Map<String, dynamic>
                : <String, dynamic>{}),
            'requestId': requestId,
          },
          'payload': callback['payload'] is Map<String, dynamic>
              ? callback['payload'] as Map<String, dynamic>
              : <String, dynamic>{},
        };
      },
    );
    if (success && mounted && _linkingTokenController.text.trim().isNotEmpty) {
      setState(() => _completedStep2Index = 1);
    }
  }

  Future<void> _callOpenConsentView() async {
    try {
      final url = _required(
        _consentViewUrlController,
        'Generated Consent View URL',
      );
      await _openExternalUrl(url);
      if (!mounted) return;
      ScaffoldMessenger.of(
        context,
      ).showSnackBar(const SnackBar(content: Text('Consent page opened')));
    } catch (e) {
      if (!mounted) return;
      await _showErrorDialog(title: 'Operation Failed', error: e);
    }
  }

  Future<void> _openExternalUrl(String url) async {
    ProcessResult result;
    if (Platform.isMacOS) {
      result = await Process.run('open', [url]);
    } else if (Platform.isLinux) {
      result = await Process.run('xdg-open', [url]);
    } else if (Platform.isWindows) {
      result = await Process.run('cmd', ['/c', 'start', '', url]);
    } else {
      throw Exception('Unsupported desktop platform for opening external URLs');
    }

    if (result.exitCode != 0) {
      throw Exception('Failed to open consent page URL');
    }
  }
